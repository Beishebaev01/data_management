WITH DailyHackers AS (
    SELECT 
        submission_date,
        hacker_id,
        COUNT(submission_id) AS sub_count
    FROM Submissions
    GROUP BY submission_date, hacker_id
),
LoyalHackers AS (
    SELECT 
        submission_date,
        hacker_id
    FROM DailyHackers dh1
    WHERE (submission_date - CAST('2016-03-01' AS DATE)) = (
        SELECT COUNT(DISTINCT submission_date)
        FROM Submissions dh2
        WHERE dh2.hacker_id = dh1.hacker_id
          AND dh2.submission_date < dh1.submission_date
    )
),
LoyalCount AS (
    SELECT 
        submission_date,
        COUNT(DISTINCT hacker_id) AS unique_loyal_hackers
    FROM LoyalHackers
    GROUP BY submission_date
),
RankedDailyHackers AS (
    SELECT 
        submission_date,
        hacker_id,
        ROW_NUMBER() OVER (
            PARTITION BY submission_date 
            ORDER BY sub_count DESC, hacker_id ASC
        ) AS rn
    FROM DailyHackers
)
SELECT 
    lc.submission_date,
    lc.unique_loyal_hackers,
    rdh.hacker_id,
    h.name
FROM LoyalCount lc
JOIN RankedDailyHackers rdh ON lc.submission_date = rdh.submission_date AND rdh.rn = 1
JOIN Hackers h ON rdh.hacker_id = h.hacker_id
ORDER BY lc.submission_date ASC;
